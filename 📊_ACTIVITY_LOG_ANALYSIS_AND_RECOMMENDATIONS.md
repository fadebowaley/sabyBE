#### 2. Improve Frontend Validation

```javascript
// File: sabyFrontend/apps/web/utils/formValidation.ts

const validateSubmission = (data) => {
  const errors = [];

  // Check month format
  if (data.month && !/^\d{4}-\d{2}-\d{2}$/.test(data.month)) {
    errors.push('Month must be in YYYY-MM-DD format');
  }

  // Check required fields
  if (!data.projectId) errors.push('Project is required');
  if (!data.formId) errors.push('Form is required');

  // Check payload
  if (!data.payload || Object.keys(data.payload).length === 0) {
    errors.push('Form data cannot be empty');
  }

  return {
    isValid: errors.length === 0,
    errors,
  };
};

// Use before submission:
const validation = validateSubmission(formData);
if (!validation.isValid) {
  // Show errors to user
  setErrors(validation.errors);
  return;
}

// If valid, proceed with submission
await submitForm(formData);
```
